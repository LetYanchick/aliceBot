cmake_minimum_required(VERSION 3.16)
project(DungeonSim CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ── Sources shared between main and tests ──────────────────────────────────
set(DUNGEON_SOURCES
    Room.cpp
    DungeonState.cpp
    Parser.cpp
    Logger.cpp
    Pathfinder.cpp
    AliceBot.cpp
)

# ── Main executable ────────────────────────────────────────────────────────
add_executable(task main.cpp ${DUNGEON_SOURCES})
target_include_directories(task PRIVATE ${CMAKE_SOURCE_DIR})

# ── Tests ──────────────────────────────────────────────────────────────────
enable_testing()

include(FetchContent)
FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        v1.14.0
)
# Don't install gtest alongside the project
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

add_executable(dungeon_tests
    tests/RoomTest.cpp
    tests/DungeonStateTest.cpp
    tests/ParserTest.cpp
    tests/PathfinderTest.cpp
    tests/AliceBotTest.cpp
    ${DUNGEON_SOURCES}
)

target_include_directories(dungeon_tests PRIVATE ${CMAKE_SOURCE_DIR})
target_link_libraries(dungeon_tests PRIVATE GTest::gtest GTest::gtest_main)

include(GoogleTest)
gtest_discover_tests(dungeon_tests)
